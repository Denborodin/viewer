cmake_minimum_required(VERSION 3.25)
project(Viewer VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Viewer requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
foreach(opt WEBP_BUILD_ANIM_UTILS WEBP_BUILD_CWEBP WEBP_BUILD_DWEBP WEBP_BUILD_GIF2WEBP WEBP_BUILD_IMG2WEBP WEBP_BUILD_VWEBP WEBP_BUILD_WEBPINFO WEBP_BUILD_WEBPMUX WEBP_BUILD_EXTRAS WEBP_BUILD_LIBWEBPMUX)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/libwebp-1.6.0 EXCLUDE_FROM_ALL)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/zlib-1.3.2 EXCLUDE_FROM_ALL)
add_library(turbojpeg SHARED IMPORTED)
set_target_properties(turbojpeg PROPERTIES
  IMPORTED_IMPLIB "${CMAKE_SOURCE_DIR}/third_party/jpeg-bin/lib/turbojpeg.lib"
  IMPORTED_LOCATION "${CMAKE_SOURCE_DIR}/third_party/jpeg-bin/bin/turbojpeg.dll"
  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/third_party/libjpeg-turbo-3.2.0/src")
add_library(viewer_core STATIC src/common.cpp src/state.cpp src/cache.cpp src/source.cpp src/archive.cpp src/decode.cpp src/psd.cpp src/pipeline.cpp)
target_include_directories(viewer_core PUBLIC src PRIVATE third_party/7zip/CPP third_party/libwebp-1.6.0/src)
target_link_libraries(viewer_core PUBLIC windowscodecs ole32 oleaut32 shlwapi shell32 bcrypt winsqlite3 webp webpdemux turbojpeg ZLIB::ZLIBSTATIC)
target_compile_options(viewer_core PRIVATE /W4)
add_executable(Viewer WIN32 src/main.cpp src/renderer.cpp src/shell_integration.cpp src/viewer.rc)
target_link_libraries(Viewer PRIVATE viewer_core d2d1 dwrite dwmapi comctl32 comdlg32 propsys windowsapp)
target_compile_options(Viewer PRIVATE /W4)
target_link_options(Viewer PRIVATE /MANIFEST:NO)
add_library(ViewerShell SHARED src/shell_extension.cpp src/common.cpp src/shell.def src/shell.rc)
target_include_directories(ViewerShell PRIVATE src)
target_link_libraries(ViewerShell PRIVATE ole32 shell32 shlwapi bcrypt)
set_target_properties(ViewerShell PROPERTIES PREFIX "")
add_executable(ViewerTests tests/tests.cpp)
target_link_libraries(ViewerTests PRIVATE viewer_core)
add_executable(ViewerJpegBench tools/jpeg_benchmark.cpp)
target_link_libraries(ViewerJpegBench PRIVATE viewer_core)
add_executable(ViewerPsdTests tests/psd_tests.cpp)
target_link_libraries(ViewerPsdTests PRIVATE viewer_core)
add_executable(ViewerJpegTests tests/jpeg_tests.cpp)
target_link_libraries(ViewerJpegTests PRIVATE viewer_core)
add_executable(ViewerBench tools/benchmark.cpp src/renderer.cpp)
target_link_libraries(ViewerBench PRIVATE viewer_core d2d1 dwrite)
add_executable(ViewerGraphicsTests tests/graphics_tests.cpp src/renderer.cpp)
target_include_directories(ViewerGraphicsTests PRIVATE third_party/libwebp-1.6.0/src)
target_link_libraries(ViewerGraphicsTests PRIVATE viewer_core d2d1 dwrite)
add_dependencies(ViewerGraphicsTests ViewerShell)
foreach(t Viewer ViewerTests ViewerBench)
  add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/7zip-bin/7z.dll" "$<TARGET_FILE_DIR:${t}>/7z.dll")
endforeach()
foreach(t Viewer ViewerTests ViewerBench ViewerGraphicsTests ViewerJpegBench ViewerJpegTests ViewerPsdTests)
  add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:turbojpeg>" "$<TARGET_FILE_DIR:${t}>/turbojpeg.dll")
endforeach()
enable_testing()
add_test(NAME viewer_core COMMAND ViewerTests "${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME viewer_graphics_shell COMMAND ViewerGraphicsTests "${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME viewer_jpeg COMMAND ViewerJpegTests "${CMAKE_SOURCE_DIR}/tests/fixtures/generated/jpeg-colors")
add_test(NAME viewer_psd COMMAND ViewerPsdTests "${CMAKE_SOURCE_DIR}/tests/fixtures/generated/psd")
set_tests_properties(viewer_core PROPERTIES TIMEOUT 120)
install(TARGETS Viewer ViewerShell ViewerBench ViewerJpegBench RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${CMAKE_SOURCE_DIR}/third_party/jpeg-bin/bin/turbojpeg.dll" DESTINATION .)
install(FILES third_party/libjpeg-turbo-3.2.0/LICENSE.md DESTINATION licenses RENAME libjpeg-turbo.md)
install(FILES third_party/libjpeg-turbo-3.2.0/README.ijg DESTINATION licenses RENAME libjpeg-ijg.txt)
install(FILES third_party/libjpeg-turbo.tar.gz DESTINATION third-party-source)
install(FILES third_party/7zip-bin/7z.dll DESTINATION .)
install(DIRECTORY packaging/ DESTINATION integration PATTERN "*.in" EXCLUDE)
install(DIRECTORY packaging/Assets DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES THIRD_PARTY.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES third_party/7zip/DOC/License.txt DESTINATION licenses RENAME 7zip.txt)
install(FILES third_party/7zip/DOC/copying.txt third_party/7zip/DOC/unRarLicense.txt DESTINATION licenses)
install(FILES third_party/libwebp-1.6.0/COPYING DESTINATION licenses RENAME libwebp.txt)
install(FILES third_party/libwebp-1.6.0/PATENTS DESTINATION licenses RENAME libwebp-PATENTS.txt)
install(FILES third_party/7zip-src.tar.xz third_party/libwebp.tar.gz DESTINATION third-party-source)

install(FILES third_party/zlib-1.3.2/LICENSE DESTINATION licenses RENAME zlib.txt)
install(FILES third_party/zlib.tar.gz DESTINATION third-party-source)
